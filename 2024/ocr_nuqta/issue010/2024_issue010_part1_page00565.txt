إن مجلس الوزراء

بناء على المرسوم الاشتراعي رقم ١٣٩ تاريخ ١٩٥٩/٦/١٢ (تنظيم المديرية العامة للأمن العام)،
بناء على المرسوم الاشتراعي رقم ١٠٢ تاريخ ١٩٨٣/٩/١٦ (قانون الدفاع الوطني)،
بناء على القانون رقم ١٧ تاريخ ١٩٩٠/٩/٦ (تنظيم قوى الأمن الداخلي)،
بناء على المرسوم رقم ١٨٠٢ تاريخ ١٩٧٩/٢/٢٧ (تنظيم الضابطة الجمركية)،
بناء على المرسوم رقم ٢٦٦١ تاريخ ١٩٨٥/٩/٣ (تنظيم المديرية العامة لأمن الدولة)،
بناء على المرسوم رقم ٥٣٩٧ تاريخ ١٩٨٢/٩/٢٠ (تحديد تعويضات العسكريين ورجال قوى الأمن الداخلي والأمن العام)،
بناء على المرسوم رقم ٨٧٣٩ تاريخ ٢٠٢٢/١/٢٨ وتعديلاته (إعطاء تعويض نقل شهري مقطوع للعسكريين في الجيش وقوى الأمن الداخلي والأمن العام وأمن الدولة
والضابطة الجمركية وشرطة مجلس النواب) لاسيما المرسوم رقم ١٠٩٥١ تاريخ ٢٠٢٢/١٢/٥ والمرسوم رقم ١١٩١٤ تاريخ ٢٠٢٣/٨/١٧،
بناء على اقتراح رئيس مجلس الوزراء المبني على مقتضيات المصلحة العامة تأمينا لحسن سير المرافق العامة،
بعد استشارة مجلس شورى الدولة (الرأي رقم ٢٠٢٣/٩٠-٢٠٢٤ تاريخ ٢٠٢٤/٢/٢٢)،
وبعد موافقة مجلس الوزراء بتاريخ ٢٠٢٤/٢/٢٨،
يرسم ما يأتي:
المادة الأولى: تُعدل المادة الأولى من المرسوم رقم ٨٧٣٩ تاريخ ٢٠٢٢/١/٢٨ المعدلة بموجب المرسوم رقم ١٠٩٥١ تاريخ ٢٠٢٢/١٢/٥ ووفقا للمرسوم رقم ١١٩١٤ تاريخ ٢٠٢٣/٨/١٧ بحيث يعطى العسكريين
في الجيش وقوى الأمن الداخلي والأمن العام وأمن الدولة والضابطة الجمركية وشرطة مجلس النواب، تعويض نقل شهري مقطوع بقيمة /٩,٠٠٠,٠٠٠/ل.ل. (فقط تسعة ملايين ليرة لبنانية) بدلا من /٥,٠٠٠,٠٠٠/ل.ل. (خمسة ملايين ليرة لبنانية)، أي بزيادة قدرها /٤,٠٠٠,٠٠٠/ ليرة لبنانية، ويمتقيد من هذا التعويض جميع العسكريين من الرتب كافة، وذلك اعتبارا من ٢٠٢٤/٣/١،
ويتقاضى، لمرة واحدة فقط، الأشخاص المُحددين أعلاه، تعويضا إضافيا قدره /١٢,٠٠٠,٠٠٠/ ل.ل. (فقط إثنا عشر مليون ليرة لبنانية).
المادة الثانية: يُنشر هذا المرسوم ويبلغ حيث تدعو الحاجة ويُعمل به فور صدوره.

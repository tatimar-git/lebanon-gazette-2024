وزارة الداخلية والبلديات
قرار رقم ١٧٧٤
تفويض موظفين في وزارة الداخلية والبلديات ببعض الصلاحيات
ان وزير الداخلية والبلديات،
بناء على المرسوم رقم ٨٣٧٦ تاريخ ٢٠٢١/٩/١٠ (تشكيل الحكومة)،
بناء على المرسوم الاشتراعي رقم ١١١ تاريخ ١٩٥٩/٦/١٢ (تنظيم الادارات العامة)،
بناء على المرسوم الاشتراعي رقم ١١٢ تاريخ ١٩٥٩/٦/١٢ (نظام الموظفين) وتعديلاته،
بناء على المرسوم رقم ٤٠٨٢ تاريخ ٢٠٠٠/١٠/١٤ (تنظيم وزارة الداخلية والبلديات) لا سيما المادة الثانية منه،
بناء على القانون رقم ٢٤٤ تاريخ ٢٠٢١/٧/١٩ (قانون الشراء العام في لبنان) لا سيما المواد ٤٧ و٦٠ و٦١ منه،
بناء على المرسوم رقم ١٤٠٦٣ تاريخ ٢٠٢٤/١٠/٣ (تعديل السقوف المالية الواردة في بعض مواد قانون الشراء العام).
بناء على القرار رقم ١٦٧٢ تاريخ ٢٠٢٤/١١/١٣،
يقرر ما يأتي:
المادة الأولى: يفوض الى المدراء العامين والمحافظين والمدراء ورؤساء الاجهزة التالية اسماؤهم، بالاضافة الى مهامهم الاساسية المنصوص عليها في القوانين والانظمة المرعية الاجراء صلاحية:
- البت بالصفقات التي تُعقد بموجب فاتورة والتي لا تتجاوز سقفا ماليا محددا ب /١,٥٠٠,٠٠٠,٠٠٠/ مليار وخمسمماية مليون ليرة لبنانية وفقا للمادة /٤٧/ من قانون الشراء العام في لبنان المعدلة بالمرسوم رقم ١٤٠٦٣ تاريخ ٢٠٢٤/١٠/٣:

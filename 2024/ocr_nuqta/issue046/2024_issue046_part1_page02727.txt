قرار رقم ٧٥٢/م/٢٠٢٤
تعديل القرار رقم ١٨٦/م/٢٠٢٣
تاريخ ٢٠٢٣/٣/١٦
(الاعتراف ببرنامج الإجازة الجامعية
في المعلوماتية في كلية الآداب والعلوم
في جامعة قينيسيا)
إن وزير التربية والتعليم العالي،
بناء على المرسوم رقم ٨٣٧٦ تاريخ ٢٠٢١/٩/١٠ (تشكيل الحكومة)،
بناء على القانون رقم ٢٨٥ تاريخ ٢٠١٤/٤/٣٠ (الأحكام العامة للتعليم العالي وتنظيم التعليم العالي الخاص)،
بناء على المرسوم رقم ٢١٧٦ تاريخ ٢٠١٨/١/١٢ (الآليات الخاصة بالترخيص وبمنح الإذن بمباشرة التدريس والاعتراف وتجديد الاعتراف بالشهادات في مؤسسات التعليم العالي الخاص)،
بناء على المرسوم رقم ٩٠٨٩ تاريخ ٢٠١٢/١٠/١٠ (الترخيص بإنشاء جامعة فينيسيا الدولية)، وعلى المرسوم رقم ١١٢٠ تاريخ ٢٠١٧/٧/٢١ (الترخيص بتعديل تسمية جامعة فينيسيا الدولية الى جامعة فينيسيا وباستحداث برنامج ماجستير في إدارة الأعمال)،
بناء على القرار رقم ٧١٦/م/٢٠١٥ تاريخ ٢٠١٥/٨/٢٠ (الترخيص لجامعة فينيسيا الدولية بمباشرة التدريس في بعض الكليات والاختصاصات)،
بناء على توصية اللجنة الفنية الأكاديمية في الجلسة رقم ٢٠٢٣/١ تاريخ ٢٠٢٣/١/٢٠، المسندة الى تقرير اللجنة
المتخصصة بالبرامج بتاريخ ٢٠٢٢/١٠/٥،
بناء على القرار رقم ١٨٦/م/٢٠٢٣ تاريخ ٢٠٢٣/٣/١٦ (الاعتراف ببرنامج الإجازة الجامعية في المعلوماتية في كلية الآداب والعلوم في جامعة فينيسيا)،
بناء على المادة ٤٩ من القانون رقم ٢٨٥ تاريخ ٢٠١٤/٤/٣٠،
بناء على الكتاب الوارد من جامعة فينيسيا تحت الرقم ٢٠٢٤/١٢٧٣ تاريخ ٢٠٢٤/٧/٣١،
